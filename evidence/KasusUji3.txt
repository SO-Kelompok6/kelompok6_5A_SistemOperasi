#KU 3.1 dan 3.2
if sha256sum -c "$S"; then echo "HASIL: LULUS"; exit 0   #Menghitung ulang hash arsip dan membandingkannya dengan nilai di berkas .sha256. Jika cocok, LULUS dengan exit status 0.
else echo "HASIL: GAGAL"; exit 1; fi   #Jika berbeda, GAGAL dengan exit status 1. Uji negatif memakai salinan arsip yang ditambah satu baris teks.
 
#KU 3.3
verify arsip asli lagi   #Memastikan arsip asli tidak ikut berubah setelah uji negatif.
 
#KU 3.4 dan 3.5
sudo -u backupop -i   #backupop anggota backupgrp sehingga boleh mengakses arsip.
cd /opt/backup-sistem   #alya bukan anggota backupgrp sehingga kernel menolak (Permission denied).
